# The bot's real .py files live in THIS repo (homelab-bot) under
# src/lib/pi-bot/  (main.py, scripts/, modules/, root/, crontab.txt,
# deploy_manifest.txt). /sync pulls the repo on the Pi, mirrors
# src/lib/pi-bot/{main.py,scripts,modules} into ~/secure-pi-bot runtime
# paths, installs root-owned files from deploy_manifest.txt, then reboots.
#
# One repo, one path. Edit scripts in the dashboard (src/lib/pi-bot/*.py),
# commit+push to GitHub, then /sync from Discord. No manual copying.

# ============================================================
# ONE-TIME BOOTSTRAP on the Pi (do this ONCE over SSH)
# ============================================================
# 1. Clone the repo as ~/secure-pi-bot (your dashboard repo -- public, no auth):
cd ~
git clone https://github.com/alon3141592653589793/homelab-bot.git secure-pi-bot
#    (if ~/secure-pi-bot already has your .env / logs, clone elsewhere then
#     move .git in:  git clone ... /tmp/hlb && mv /tmp/hlb/.git ~/secure-pi-bot/
#     and  cp -r /tmp/hlb/src ~/secure-pi-bot/)

# 2. Restore your live .env if the clone didn't include it (it's gitignored):
#    nano ~/secure-pi-bot/.env   # DISCORD_BOT_TOKEN, ALLOWED_USER_ID, channel IDs...

# 3. Install the root helper (pi_deploy_root.sh) ONCE so /sync can install
#    root-owned files later. It's in the repo under src/lib/pi-bot/root/:
sudo cp ~/secure-pi-bot/src/lib/pi-bot/root/pi_deploy_root.sh /usr/local/bin/pi_deploy_root
sudo chmod 755 /usr/local/bin/pi_deploy_root
echo "alon ALL=(root) NOPASSWD: /usr/local/bin/pi_deploy_root" | sudo tee /etc/sudoers.d/pi-deploy
sudo chmod 440 /etc/sudoers.d/pi-deploy && sudo visudo -c

# 4. Run the FIRST /sync by hand (the bot's pi_deploy.py isn't live yet):
python3 ~/secure-pi-bot/src/lib/pi-bot/scripts/pi_deploy.py --dry-run   # show what would change
python3 ~/secure-pi-bot/src/lib/pi-bot/scripts/pi_deploy.py --no-reboot # mirror + install, stay up
#    After this ~/secure-pi-bot/{main.py,scripts/,modules/} are populated and
#    /usr/local/bin/{pi-maintenance.sh,pi-audit.sh,led_ctl,...} are installed.
#    Start the bot, then from Discord:  /sync   (pulls, mirrors, reboots).

# ============================================================
# POLKIT RULE — allow alon to reboot/poweroff without sudo
# ============================================================
sudo tee /etc/polkit-1/rules.d/49-pi-bot.rules << 'POLKIT'
polkit.addRule(function(action, subject) {
    if ((action.id == "org.freedesktop.login1.reboot" ||
         action.id == "org.freedesktop.login1.power-off" ||
         action.id == "org.freedesktop.login1.reboot-multiple-sessions" ||
         action.id == "org.freedesktop.login1.power-off-multiple-sessions") &&
        subject.user == "alon") {
        return polkit.Result.YES;
    }
});
POLKIT
sudo systemctl restart polkit
